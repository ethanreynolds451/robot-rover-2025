From sensor board: 

// t for each sensor is offset from packet timestamp
// calculated on packet construction, may be off by +- a few ms
// n is sensor index
// s is sensor status, first hex digit is state and second is error 
{t[arduino_timestamp]
mpu[n, t, a:accel_x;y;z, g:gyro_x;y;z, tp:temp]
qmc[n, t, h:head, m:mag_x;y;z]
gps[n, t, s, p:fix_sign;lat;sign;long, a:sign;alt, v:vel, h:head, g:time, f:n_sats]
tof[n, t, d:dist, l:light]
us[n, t, d:dist]
tmp[n, t, tp:temp]
str[t, c:calibrated, p:pos, d:dir]
ir[n, t, s, a:addr, c:cmd, d:data]}

// Example:
{t[1234]
mpu[1,10,123:123:123,234:234:234,30]
qmc[1,15, mag, head, temp]
gps[n:0,t:1234,p;0;9A6B327C;1;764B7DE5,a:0;927B343F,v:27CD648A,h:82F41AD9,g:29A62FF3,f:03]
tof[n, t, dist, light]us[n, dist]
tmp[n, t, temp]
str[t, dir, pos, spd]
ir[n:0,t:1234,s:40,a:8E30,c:7AF3,d:B74C96A1]}
