From sensor board: 

// t for each sensor is offset from packet timestamp
// calculated on packet construction, may be off by +- a few ms
// n is sensor index
// s is sensor status, first hex digit is state and second is error 
{t[arduino_timestamp]
mpu[n, t, s, a:accel_x;y;z, g:gyro_x;y;z, c:temp]
qmc[n, t, h:head, m:mag_x;y;z]
gps[n, t, s, p:lat;long, a:alt, v:vel, h:head, g:time, f:n_sats]
tof[n, t, d:dist, l:light]
us[n, t, d:dist]
tmp[n, t, tp:temp]
enc[n, t, s, p:sign;pos]
ir[n, t, s, a:addr, c:cmd, d:data]}

// Examples (each value represented is max possible length of chars)
{t[1234ABCD]
mpu[n:0,t:1234;s:40,a:320D;1B23;F398,g:32B2;A64B;95BC,c:372B]
qmc[1,15, mag, head, temp]
gps[n:0,t:1234,s:40,p;9A6B327C;764B7DE5,a:927B343F,v:27CD648A,h:82F41AD9,g:29A62FF3,f:03]
tof[n, t, dist, light]us[n, dist]
tmp[n, t, temp]
enc[n:0,t:1234,s:40,p:902C43B2]
ir[n:0,t:1234,s:40,a:8E30,c:7AF3,d:B74C96A1]}
